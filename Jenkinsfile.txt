// bankapp application pipeline (declarative).
//
//  Checkout -> SonarQube + Quality Gate -> OWASP Dependency-Check ->
//  Build & Unit Test -> Publish JAR to Nexus -> Build Docker image ->
//  Trivy scan -> Push image to Nexus ->
//  Deploy to Stage -> [manual approval] -> Deploy to Prod.
//
// Put this file at the ROOT of the bankapp APP repo as `Jenkinsfile`.
// Terraform pre-creates the `bankapp` pipeline job pointing at that repo, and
// injects ANSIBLE_HOST / RDS_ENDPOINT / DB_NAME / NEXUS_DOCKER_REGISTRY as
// Jenkins GLOBAL env vars via JCasC - so there is nothing to hand-edit here.
//
// Credentials/config are all created by JCasC:
//   - SonarQube server 'sonarqube'        + credential 'sonar-token'
//   - Managed Maven settings 'nexus-maven-settings' (server 'nexus' -> nexus-cred)
//   - Username/password 'nexus-cred'      (Nexus CI user: docker push + mvn deploy)
//   - Username/password 'bankapp-db'      (RDS credentials)
//   - SSH private key   'ansible-key'     (ec2-user@ansible-server)

pipeline {
    agent any

    tools {
        maven 'maven'
        jdk 'java'
    }

    options {
        timestamps()
        disableConcurrentBuilds()
        buildDiscarder(logRotator(numToKeepStr: '20'))
        timeout(time: 45, unit: 'MINUTES')
    }

    environment {
        SONARQUBE_ENV     = 'sonarqube'
        MAVEN_SETTINGS    = 'nexus-maven-settings'
        NEXUS_DOCKER_CREDS = 'nexus-cred'
        IMAGE_NAME        = 'bankapp'
        TRIVY_SEVERITY    = 'HIGH,CRITICAL'
        ANSIBLE_SSH_CREDS = 'ansible-key'
        DB_CREDS          = 'bankapp-db'
        // ANSIBLE_HOST / RDS_ENDPOINT / DB_NAME / NEXUS_DOCKER_REGISTRY come from
        // the Jenkins global environment (set by Terraform/JCasC).
    }

    stages {
        stage('Preflight') {
            steps {
                script {
                    for (v in ['ANSIBLE_HOST', 'RDS_ENDPOINT', 'DB_NAME', 'NEXUS_DOCKER_REGISTRY']) {
                        if (!env.getProperty(v)) { error "Global env ${v} is not set - check JCasC / terraform apply" }
                    }
                    echo "ansible=${env.ANSIBLE_HOST}  rds=${env.RDS_ENDPOINT}  registry=${env.NEXUS_DOCKER_REGISTRY}"
                }
            }
        }

        stage('Checkout') {
            steps {
                checkout scm
                script {
                    env.APP_VERSION = sh(
                        returnStdout: true,
                        script: "awk '/<artifactId>bankapp<\\/artifactId>/{f=1} f && /<version>/{gsub(/.*<version>|<\\/version>.*/, \"\"); print; exit}' pom.xml"
                    ).trim()
                    env.IMAGE_TAG = "${env.APP_VERSION}-${env.BUILD_NUMBER}"
                    echo "version=${env.APP_VERSION}  image=${env.IMAGE_NAME}:${env.IMAGE_TAG}"
                }
            }
        }

        stage('SonarQube Analysis') {
            steps {
                catchError(buildResult: 'SUCCESS', stageResult: 'UNSTABLE') {
                    configFileProvider([configFile(fileId: "${MAVEN_SETTINGS}", variable: 'SETTINGS')]) {
                        withSonarQubeEnv("${SONARQUBE_ENV}") {
                            sh 'mvn -B -s $SETTINGS sonar:sonar -Dsonar.projectVersion=$APP_VERSION'
                        }
                    }
                }
            }
        }

        stage('Quality Gate') {
            steps {
                catchError(buildResult: 'SUCCESS', stageResult: 'UNSTABLE') {
                    timeout(time: 10, unit: 'MINUTES') {
                        // non-blocking: Sonar issues mark the stage UNSTABLE but do
                        // not stop a deploy. Flip to abortPipeline: true to enforce.
                        waitForQualityGate abortPipeline: false
                    }
                }
            }
        }

        stage('OWASP Dependency-Check') {
            steps {
                catchError(buildResult: 'SUCCESS', stageResult: 'UNSTABLE') {
                    configFileProvider([configFile(fileId: "${MAVEN_SETTINGS}", variable: 'SETTINGS')]) {
                        sh 'mvn -B -s $SETTINGS dependency-check:check ${NVD_API_KEY:+-Dnvd.api.key=$NVD_API_KEY}'
                    }
                }
            }
            post {
                always {
                    archiveArtifacts artifacts: 'target/dependency-check-report.*', allowEmptyArchive: true
                    publishHTML(target: [reportDir: 'target', reportFiles: 'dependency-check-report.html',
                                         reportName: 'OWASP Dependency-Check', keepAll: true, alwaysLinkToLastBuild: true])
                }
            }
        }
        stage('Build & Unit Test') {
            steps {
                configFileProvider([configFile(fileId: "${MAVEN_SETTINGS}", variable: 'SETTINGS')]) {
                    sh 'mvn -B -s $SETTINGS clean verify'
                }
            }
            post {
                always {
                    junit testResults: 'target/surefire-reports/*.xml', allowEmptyResults: true
                    jacoco execPattern: 'target/jacoco.exec'
                }
            }
        }
        stage('Publish JAR to Nexus') {
            steps {
                configFileProvider([configFile(fileId: "${MAVEN_SETTINGS}", variable: 'SETTINGS')]) {
                    sh 'mvn -B -s $SETTINGS deploy -DskipTests'
                }
            }
        }

        stage('Build Docker image') {
            steps {
                sh 'docker build --build-arg JAR_FILE=target/bankapp.jar -t $IMAGE_NAME:$IMAGE_TAG .'
            }
        }

        stage('Trivy scan') {
            steps {
                // Report-only (--exit-code 0) until the app bumps its vulnerable
                // deps (Spring Boot 3.3.4 -> current, Tomcat, Jackson). Switch
                // back to --exit-code 1 to gate the build once those are clean.
                sh '''
                    trivy image --scanners vuln --no-progress --ignore-unfixed \
                      --severity $TRIVY_SEVERITY --exit-code 0 \
                      $IMAGE_NAME:$IMAGE_TAG
                '''
            }
        }

        stage('Push image to Nexus') {
            steps {
                script {
                    docker.withRegistry("https://${env.NEXUS_DOCKER_REGISTRY}", env.NEXUS_DOCKER_CREDS) {
                        sh 'docker tag  $IMAGE_NAME:$IMAGE_TAG $NEXUS_DOCKER_REGISTRY/$IMAGE_NAME:$IMAGE_TAG'
                        sh 'docker push $NEXUS_DOCKER_REGISTRY/$IMAGE_NAME:$IMAGE_TAG'
                        sh 'docker tag  $IMAGE_NAME:$IMAGE_TAG $NEXUS_DOCKER_REGISTRY/$IMAGE_NAME:latest'
                        sh 'docker push $NEXUS_DOCKER_REGISTRY/$IMAGE_NAME:latest'
                    }
                }
            }
        }

        stage('Deploy to Stage') {
            steps {
                sshagent([env.ANSIBLE_SSH_CREDS]) {
                    withCredentials([usernamePassword(credentialsId: env.NEXUS_DOCKER_CREDS,
                                                      usernameVariable: 'NX_USER', passwordVariable: 'NX_PASS'),
                                     usernamePassword(credentialsId: env.DB_CREDS,
                                                      usernameVariable: 'DB_USER', passwordVariable: 'DB_PASS')]) {
                        sh '''
                            set +x
                            ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null ec2-user@"$ANSIBLE_HOST" \
                              "ansible-playbook /opt/docker/deploy-stage.yml \
                                 -e registry=$NEXUS_DOCKER_REGISTRY \
                                 -e image_ref=$NEXUS_DOCKER_REGISTRY/$IMAGE_NAME:$IMAGE_TAG \
                                 -e nexus_user='$NX_USER' -e nexus_pass='$NX_PASS' \
                                 -e db_url='jdbc:mysql://$RDS_ENDPOINT/$DB_NAME' \
                                 -e db_user='$DB_USER' -e db_pass='$DB_PASS'"
                        '''
                    }
                }
            }
        }

        stage('Approve production deployment') {
            steps {
                timeout(time: 1, unit: 'HOURS') {
                    input message: "Deploy ${env.IMAGE_NAME}:${env.IMAGE_TAG} to PRODUCTION?", ok: 'Deploy to prod'
                }
            }
        }

        stage('Deploy to Prod') {
            steps {
                sshagent([env.ANSIBLE_SSH_CREDS]) {
                    withCredentials([usernamePassword(credentialsId: env.NEXUS_DOCKER_CREDS,
                                                      usernameVariable: 'NX_USER', passwordVariable: 'NX_PASS'),
                                     usernamePassword(credentialsId: env.DB_CREDS,
                                                      usernameVariable: 'DB_USER', passwordVariable: 'DB_PASS')]) {
                        sh '''
                            set +x
                            ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null ec2-user@"$ANSIBLE_HOST" \
                              "ansible-playbook /opt/docker/deploy-prod.yml \
                                 -e registry=$NEXUS_DOCKER_REGISTRY \
                                 -e image_ref=$NEXUS_DOCKER_REGISTRY/$IMAGE_NAME:$IMAGE_TAG \
                                 -e nexus_user='$NX_USER' -e nexus_pass='$NX_PASS' \
                                 -e db_url='jdbc:mysql://$RDS_ENDPOINT/$DB_NAME' \
                                 -e db_user='$DB_USER' -e db_pass='$DB_PASS'"
                        '''
                    }
                }
            }
        }
    }

    post {
        success { echo "OK - deployed ${env.NEXUS_DOCKER_REGISTRY}/${env.IMAGE_NAME}:${env.IMAGE_TAG} to stage and prod" }
        failure { echo 'Pipeline failed - see stage logs' }
        always  { cleanWs() }
    }
}
